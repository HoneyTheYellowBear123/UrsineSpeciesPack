honeybear_ursinespeciespack_berserker_rage = {
	length = @EdictPerpetual
	icon = "GFX_edict_type_policy"

	resources = {
		category = edicts
		cost = {
			food = @Edict1Cost
			multiplier = value:edict_size_effect
		}
		upkeep = {

			multiplier = value:edict_size_effect
		}
	}

	potential = {
		OR = {
			has_resource = { type = food amount > 0 }
		}
		has_civic = civic_honeybear_ursinespeciespack_berserkers

	    army_damage_mult = 0.15

		modifier = {

			is_at_war = yes
			has_resource = { type = food amount > 1000 }
		}
	}
}
